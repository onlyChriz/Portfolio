167bool ReplicationManager::ClientConnection(INetworkTransfer::IncomingPacket in)
168{
169 ByteReader r(in.Message);
170 ConnectPayload connectRequestee;
171
172 if (!connectRequestee.Deserilize(r))
173 {
174 LOG_FAIL("Failed to deserialize ConnectMessage | Peer ID: %u", in.FromPeer);
175 return true;
176 }
177
178 Tga::Vector3f position = connectRequestee.Position;
179 ServerCoordinator::ServerPeerId resolvedEndpointId;
180
181 if (!myServerWorldSpaceResolver.ResolveServerIdFromPosition(position, resolvedEndpointId))
182 {
183 LOG_FAIL("Could not Resolve server endpoint from position for peer: %u | endpoint ID: %u", in.FromPeer,
184 resolvedEndpointId);
185 return true;
186 }
187
188 if (myCoordinator.HasEndpoint(resolvedEndpointId))
189 {
190 if (!myCoordinator.RegisterEntity(in.FromPeer, resolvedEndpointId))
191 {
192 LOG_FAIL("Could not register Entity from Peer: %u | endpoint ID: %u", in.FromPeer, resolvedEndpointId);
193 return true;
194 }
195
196 if (ServerCoordinator::ServerPeerId id = myCoordinator.GetOwner(in.FromPeer))
197 {
198 if (id == 0)
199 {
200 LOG_ERROR("ServerID from Coordinator is 0 | Entity ID: %u", in.FromPeer);
201 return true;
202 }
203
204 NetworkMessage msg(MessageType::Client_Connect);
205 ByteWriter w(msg);
206 ReplicationState connect;
207 connect.EntityId = in.FromPeer;
208 connect.Position = connectRequestee.Position;
209 connect.Scale = connectRequestee.Scale;
210 //connect.Transform = connectRequestee.Transform;
211
212 connect.AddFlag(RF_Authoritive);
213 if (!connect.Serilize(w))
214 {
215 LOG_FAIL("Failed to serialize Client Server_Connect | Peer ID: %u", in.FromPeer);
216 return true;
217 }
218 msg.payloadSize = w.GetWrittenBytes();
219
220 myNetTransferBuss->Send(msg, id);
221 }
222 }
223 return false;

274void ReplicationManager::ConsumeIncomming()
275{
276 while (myNetTransferBuss->GetIncomingCount() > 0)
277 {
278 IncomingHandleT<INetworkTransfer> handle = myNetTransferBuss->GetNextIncoming();
279 if (!handle.IsValid())
280 {
281 LOG_ERROR("Invalid message handler");
282 continue;
283 }
284
285 MessageType type;
286 if (!handle.TryPeek(type))
287 {
288 LOG_ERROR("Could not Peek at myType");
289 continue;
290 }
291 INetworkTransfer::IncomingPacket in;
292 switch (type)
293 {
294 case MessageType::Handshake:
295 {
296 if (!handle.TryConsume(in))
297 {
298 LOG_ERROR("Could not consume Message from Peer: %u", in.FromPeer);
299 continue;
300 }
301 if (in.Message.HasFlag(MessageFlags_FromClient))
302 {
303 bool foundID = false;
304 for (uint32_t handshakeID : myEntities)
305 {
306 if (handshakeID == in.FromPeer)
307 {
308 LOG_ERROR("Second handshake from Client PeerID: %u", in.FromPeer);
309 foundID = true;
310 break;
311 }
312 }
313 if (!foundID)
314 {
315 myEntities.push_back(in.FromPeer);
316 }
317 NetworkMessage msg(MessageType::Server_Accept);
318 msg.AddFlag(MessageFlags_HandshakeAck);
319 ByteWriter w(msg);
320 MultiCastInitPayload payload;
321
322 payload.MultiCastIP = myMultiCastIP;
323 payload.Serilize(w);
324 w.Finalize();
325
326 myNetTransferBuss->Send(msg, in.FromPeer);
327 }
328 break;
329 }
330 case MessageType::Server_Connect:
331 {
332 if (!handle.TryConsume(in))
333 {
334 LOG_ERROR("Could not consume Message from PeerID: %u", in.FromPeer);
335 continue;
336 }
337
338 if (in.Message.HasFlag(MessageFlags_FromServer))
339 {
340 if (ServerConnection(in))
341 continue;
342 }
343 break;
344 }
345 case MessageType::Client_Connect:
346 {
347 if (!handle.TryConsume(in))
348 {
349 LOG_ERROR("Could not consume Message from PeerID: %u", in.FromPeer);
350 continue;
351 }
352
353 if (in.Message.HasFlag(MessageFlags_FromClient))
354 {
355 if (ClientConnection(in))
356 break;
357 }
358 break;
359 }
360
361 case MessageType::Movement:
362 {
363 if (!handle.TryConsume(in))
364 {
365 LOG_ERROR("Could not consume Message from Peer: %u", in.FromPeer);
366 continue;
367 }
368
369 if (in.Message.HasFlag(MessageFlags_FromClient))
370 {
371 if (ClientToServerMovementMessage(in))
372 break;
373 }
374 if (in.Message.HasFlag(MessageFlags_FromServer))
375 {
376 PositionAndTimestamp move;
377 ByteReader r(in.Message);
378 if (!move.Deserilize(r))
379 {
380 LOG_ERROR("Could not Deserialize move message from EntityId %u", in.FromPeer);
381 break;
382 }
383 NetworkMessage msg(MessageType::Movement);
384 ByteWriter w(msg);
385 PositionAndTimestamp payload(move);
386 if (!PayloadRegistry::GetInstance().TrySerialize(payload, w))
387 {
388 LOG_ERROR("Failed to serilize PositionAndTimestamp from ServerId %u", in.FromPeer);
389 break;
390 }
391 myNetTransferBuss->Send(in.Message);
392 }
393 break;
394 }
395 case MessageType::Server_Accept:
396 {
397 if (!handle.TryConsume(in))
398 {
399 LOG_FAIL("Could not consume Message from Peer: %u", in.FromPeer);
400 continue;
401 }
402
403 if (in.Message.HasFlag(MessageFlags_FromServer))
404 {
405 ByteReader r(in.Message);
406 ReplicationState clientOwnerState;
407 if (!clientOwnerState.Deserilize(r))
408 {
409 LOG_FAIL("Failed to Deserialize Server_Accept | Peer: %u", in.FromPeer);
410 break;
411 }
412
413 if (clientOwnerState.EntityId == 0)
414 {
415 LOG_ERROR("Server_Accept target is 0, dropping");
416 break;
417 }
418
419 myNetTransferBuss->Send(in.Message, clientOwnerState.EntityId);
420 }
421 break;
422 }
423 case MessageType::SyncRequest:
424 {
425 if (!handle.TryConsume(in))
426 {
427 LOG_ERROR("Could not consume Message from Peer: %u", in.FromPeer);
428 continue;
429 }
430
431 if (in.Message.HasFlag(MessageFlags_FromClient))
432 {
433 NetworkMessage msg(MessageType::SyncRequest);
434 ByteWriter w(msg);
435 SyncRequestPayload sync;
436 sync.EntityId = in.FromPeer;
437 if (!sync.Serilize(w))
438 {
439 LOG_ERROR("Could not serilize Message sync message");

441 }

443 {
444 w.Finalize();

447 }

449 }
450 case MessageType::Client_Accept:
451 {
452 if (!handle.TryConsume(in))
453 {

455 continue;
456 }
457
458 if (in.Message.HasFlag(MessageFlags_FromServer))

469
470 myNetTransferBuss->Send(in.Message, inState.EntityId);
471 }
472 break;
473 }
474 case MessageType::LOG:
475 if (!handle.TryConsume(in))
476 {
477 LOG_INFO("Could not consume Message from Peer: %u", in.FromPeer);
478 continue;
479 }
480 LOG_INFO(in.Message.messageBuffer);
481 break;
482 case MessageType::Error:
483 {
484 if (!handle.TryConsume(in))
485 {
486 LOG_FAIL("Could not consume Message from Peer: %u", in.FromPeer);
487 continue;
488 }
489 LOG_ERROR(in.Message.messageBuffer);
490 break;
491 }
492 case MessageType::SyncUpdate:
493 {
494 if (!handle.TryConsume(in))
495 {
496 LOG_ERROR("Could not consume Message from Peer: %u", in.FromPeer);
497 continue;
498 }
499
500 if (in.Message.HasFlag(MessageFlags_MultiMessage) &&
501 in.Message.HasFlag(MessageFlags_FromServer))
502 {
503 myNetTransferBuss->Send(in.Message);
504 }
505 break;
506 }
507 case MessageType::OwnershipTransfer:
508 {
509 if (!handle.TryConsume(in))
510 {
511 LOG_ERROR("Could not consume Message from Peer: %u", in.FromPeer);
512 continue;
513 }
514
515 TransferRequest request;
516 ByteReader r(in.Message);
517 if (!request.Deserilize(r))
518 {
519 LOG_FAIL("Failed to Deserialize TransferRequest | from Peer ID: %u", in.FromPeer);
520 break;
521 }
522 auto it = myActiveTransfers.find(request.EntityID);
523 if (it != myActiveTransfers.end())
524 {
525 const HandoffToken& token = it->second;
526
527 if (token.Phase != TransferPhase::Done &&
528 token.Phase != TransferPhase::Failed)
529 {
530 continue;
531 }
532 }
533
534 TransferDecision decision;
535 if (myServerWorldSpaceResolver.ResolveServerTransferRequest(request, decision))
536 {
537 StartTransfer(decision);
538 }
539 break;
540 }
541 case MessageType::OwnershipAck:
542 {
543 break;
544 }
545 case MessageType::MultiPartMessage:
546 case MessageType::Unknown:
547 case MessageType::Disconnect:
548 default:
549 handle.Skip();
550 break;
551 }
552 }
