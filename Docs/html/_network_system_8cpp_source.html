428PeerId NetworkSystem::ResolveOrRegisterPeer(const sockaddr_in& aFrom)
429{
430 uint64_t key = ((uint64_t)aFrom.sin_addr.s_addr << 32) | ntohs(aFrom.sin_port);
431
432 // Check old map first (for compatibility)
433 auto addrIt = myAddrToPeer.find(key);
434 if (addrIt != myAddrToPeer.end())
435 {
436 PeerId id = addrIt->second;
437 // Update last seen in new structure
438 auto peerIt = myInternalPeers.find(id);
439 if (peerIt != myInternalPeers.end())
440 {
441 peerIt->second.myLastSeenTime = NowMS();
442 }
443 return id;
444 }
445
446 // New peer
447 PeerId newId = myNextPeerId++;
448
449 // New internal storage
450 PeerInternal internal{};
451 internal.myId = newId;
452 internal.myAddress = aFrom;
453 internal.myLastSeenTime = NowMS();
454 myInternalPeers.emplace(newId, std::move(internal));
455
456 // New game info
457 PeerInfo info{};
458 info.myId = newId;
459 myPeerGameInfo.emplace(newId, std::move(info));
460
461 // Keep old maps for compatibility (bridge)
462 myAddrToPeer[key] = newId;
463 myPeerToAddr[newId] = aFrom;
464
465 return newId;
